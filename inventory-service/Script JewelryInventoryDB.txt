/* =========================================================
   STEP 1: DROP EXISTING DEMO TABLE
   WARNING: This deletes existing inventory_items data.
   ========================================================= */

DROP TABLE IF EXISTS dbo.inventory_items;
GO

/* =========================================================
   STEP 2: CREATE INVENTORY ITEMS TABLE
   One row represents one identifiable physical item.
   ========================================================= */

CREATE TABLE dbo.inventory_items
(
    inventory_item_id BIGINT IDENTITY(1,1) NOT NULL,
    product_id INT NOT NULL,

    barcode VARCHAR(50) NOT NULL,
    serial_number VARCHAR(100) NULL,

    actual_weight_grams DECIMAL(10,3) NULL,
    actual_gold_purity_percent DECIMAL(5,2) NULL,
    certificate_number VARCHAR(100) NULL,

    location_code VARCHAR(50) NULL,

    status VARCHAR(20) NOT NULL
        CONSTRAINT DF_inventory_items_status
        DEFAULT ('AVAILABLE'),

    reservation_reference VARCHAR(100) NULL,
    reserved_until DATETIME2(3) NULL,

    created_at DATETIME2(3) NOT NULL
        CONSTRAINT DF_inventory_items_created_at
        DEFAULT (SYSUTCDATETIME()),

    updated_at DATETIME2(3) NOT NULL
        CONSTRAINT DF_inventory_items_updated_at
        DEFAULT (SYSUTCDATETIME()),

    row_version ROWVERSION NOT NULL,

    /* Primary key */
    CONSTRAINT PK_inventory_items
        PRIMARY KEY (inventory_item_id),

    /* Each barcode identifies one physical item */
    CONSTRAINT UQ_inventory_items_barcode
        UNIQUE (barcode),

    /* A valid product reference must be positive */
    CONSTRAINT CK_inventory_items_product_id
        CHECK (product_id > 0),

    /* Weight must be positive when supplied */
    CONSTRAINT CK_inventory_items_weight
        CHECK (
            actual_weight_grams IS NULL
            OR actual_weight_grams > 0
        ),

    /* Gold purity is a percentage between 0 and 100 */
    CONSTRAINT CK_inventory_items_gold_purity
        CHECK (
            actual_gold_purity_percent IS NULL
            OR actual_gold_purity_percent
               BETWEEN 0.01 AND 100.00
        ),

    /* Only the supported inventory statuses are allowed */
    CONSTRAINT CK_inventory_items_status
        CHECK (
            status IN (
                'AVAILABLE',
                'RESERVED',
                'SOLD',
                'DAMAGED'
            )
        ),

    /* Reservation fields must agree with the item status */
    CONSTRAINT CK_inventory_items_reservation
        CHECK (
            (
                status = 'RESERVED'
                AND reservation_reference IS NOT NULL
                AND reserved_until IS NOT NULL
            )
            OR
            (
                status <> 'RESERVED'
                AND reservation_reference IS NULL
                AND reserved_until IS NULL
            )
        )
);
GO

/* =========================================================
   STEP 3: CREATE INDEXES
   ========================================================= */

-- Serial numbers are unique when provided.
-- Multiple rows may have serial_number = NULL.
CREATE UNIQUE INDEX UX_inventory_items_serial_number
    ON dbo.inventory_items(serial_number)
    WHERE serial_number IS NOT NULL;
GO

-- Find inventory items belonging to a product and status.
CREATE INDEX IX_inventory_items_product_status
    ON dbo.inventory_items(product_id, status);
GO

-- Find reservations by expiration time.
CREATE INDEX IX_inventory_items_reserved_until
    ON dbo.inventory_items(reserved_until)
    WHERE status = 'RESERVED';
GO
